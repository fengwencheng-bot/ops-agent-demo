-- update_table: Daily order data aggregation
-- Owner: bryann.yeapkk@shopee.com
 
INSERT OVERWRITE TABLE dwd.order_detail
PARTITION (dt = '${bizdate}')
SELECT
o.order_id,
o.user_id,
p.product_name,
p.category,
o.amount,
o.currency,
u.country,
u.register_date,
COLLECT_LIST(STRUCT(o.item_id, o.quantity, o.unit_price)) AS items, -- Data skew risk: high-volume sellers
o.create_time,
o.update_time
FROM ods.raw_order_data o
JOIN ods.product_catalog p ON o.product_id = p.product_id -- Full JOIN, table p is 2.8GB
JOIN dim.user_info u ON o.user_id = u.user_id
WHERE o.dt = '${bizdate}'
AND o.status IN ('COMPLETED', 'REFUNDED')
DISTRIBUTE BY o.user_id

What does its code do
This task runs Spark SQL. Core logic:

1. Read the day’s orders from ods.raw_order_data
2. JOIN ods.product_catalog
3. JOIN dim.user_info
4. COLLECT_LIST aggregates order lines per user
5. Write to the dwd.order_detail partition

The bottleneck is mainly the Shuffle caused by COLLECT_LIST and wide-table JOINs.
